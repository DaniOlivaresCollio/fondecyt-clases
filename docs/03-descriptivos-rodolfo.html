# --- 2) Función auxiliar: conteo no ponderado + ponderado (con SE e IC95%)
count_unw_w <- function(data, design, var, drop_na = TRUE) {
  # Conteo no ponderado
  unwt <- data %>%
    { if (drop_na) filter(., !is.na({{var}})) else . } %>%
    count(level = {{var}}, name = "n")

  # Conteo ponderado (total estimado + SE + IC95%)
  wt <- design %>%
    { if (drop_na) filter(., !is.na({{var}})) else . } %>%
    group_by(level = {{var}}) %>%
    summarise(
      n_w = survey_total(vartype = c("se", "ci"), level = 0.95, na.rm = TRUE),
      .groups = "drop"
    )

  # Unir y ordenar
  out <- unwt %>%
    right_join(wt, by = "level") %>%
    arrange(level)

  # Fila de total
  tot <- tibble(
    level   = "Total",
    n       = sum(out$n, na.rm = TRUE),
    n_w     = sum(out$n_w, na.rm = TRUE),
    n_w_se  = NA_real_,
    n_w_low = NA_real_,
    n_w_upp = NA_real_
  )

  bind_rows(out, tot)
}

# Helpers de formato
fmt_int <- function(x) {
  ifelse(is.na(x), "",
         formatC(round(x), format = "f", digits = 0,
                 big.mark = "", decimal.mark = "."))
}

fmt_num <- function(x, d = 2) {
  ifelse(is.na(x), "",
         formatC(x, format = "f", digits = d,
                 big.mark = "", decimal.mark = "."))
}

# --- 4) Formato nice para exportar (Kable)
fmt_table <- function(tbl, title = "", digits_se = 2, digits_ci = 2) {
  tbl %>%
    mutate(
      n       = fmt_int(n),
      n_w     = fmt_int(n_w),
      n_w_se  = ifelse(is.na(n_w_se),  "", fmt_num(n_w_se,  digits_se)),
      n_w_low = ifelse(is.na(n_w_low), "", fmt_num(n_w_low, digits_ci)),
      n_w_upp = ifelse(is.na(n_w_upp), "", fmt_num(n_w_upp, digits_ci))
    ) %>%
    rename(
      "Categoría"         = level,
      "n (sin ponderar)"  = n,
      "n ponderado"       = n_w,
      "SE (ponderado)"    = n_w_se,
      "IC95% inferior"    = n_w_low,
      "IC95% superior"    = n_w_upp
    ) %>%
    kbl(caption = title, align = c("l","r","r","r","r","r"), booktabs = TRUE) %>%
    kable_classic(full_width = FALSE) %>%
    add_header_above(c(" " = 1, "Conteos" = 2, "Incertidumbre (ponderado)" = 3))
}

# helper para escribir una tabla "cruda" (no la formateada a texto),
# con estilos: enteros en n/n_w y decimales en SE/IC
write_sheet_counts <- function(wb, sheet_name, df) {
  addWorksheet(wb, sheet_name)

  # Asegurar orden y tipos
  df_out <- df %>%
    select(level, n, n_w, n_w_se, n_w_low, n_w_upp)

  writeData(wb, sheet_name, df_out)

  # Estilos
  style_header <- createStyle(textDecoration = "bold")
  addStyle(wb, sheet = sheet_name, style = style_header, rows = 1, cols = 1:ncol(df_out), gridExpand = TRUE)

  # Formato sin miles para enteros (0 decimales)
  style_int <- createStyle(numFmt = "0")
  # Formato con 2 decimales para SE e IC
  style_num2 <- createStyle(numFmt = "0.00")

  n_rows <- nrow(df_out) + 1  # +1 por encabezado

  # columnas
  col_level  <- which(names(df_out) == "level")
  col_n      <- which(names(df_out) == "n")
  col_nw     <- which(names(df_out) == "n_w")
  col_se     <- which(names(df_out) == "n_w_se")
  col_low    <- which(names(df_out) == "n_w_low")
  col_upp    <- which(names(df_out) == "n_w_upp")

  # Aplicar estilos numéricos (solo a columnas numéricas, filas de datos)
  if (!is.null(col_n))  addStyle(wb, sheet_name, style_int,  rows = 2:n_rows, cols = col_n,  gridExpand = TRUE, stack = TRUE)
  if (!is.null(col_nw)) addStyle(wb, sheet_name, style_int,  rows = 2:n_rows, cols = col_nw, gridExpand = TRUE, stack = TRUE)
  if (!is.null(col_se)) addStyle(wb, sheet_name, style_num2, rows = 2:n_rows, cols = col_se, gridExpand = TRUE, stack = TRUE)
  if (!is.null(col_low))addStyle(wb, sheet_name, style_num2, rows = 2:n_rows, cols = col_low,gridExpand = TRUE, stack = TRUE)
  if (!is.null(col_upp))addStyle(wb, sheet_name, style_num2, rows = 2:n_rows, cols = col_upp,gridExpand = TRUE, stack = TRUE)

  setColWidths(wb, sheet = sheet_name, cols = 1:ncol(df_out), widths = "auto")
}

# 9) Chequeos de coherencia (recomendado)
#    Suma ponderada por categorías ≈ suma de pesos (sin NA en la variable)
# =========================================================
check_sum <- function(tbl, var_name) {
  suma_tabla <- tbl %>% filter(Categoría != "Total") %>%
    mutate(n_w = as.numeric(gsub(",", ".", gsub("\\.", "", `n ponderado`)))) %>% # por si llega formateado
    summarise(suma = sum(n_w, na.rm = TRUE)) %>% pull(suma)

  suma_pesos <- data %>% filter(!is.na(.data[[var_name]])) %>%
    summarise(suma = sum(rake_wb2, na.rm = TRUE)) %>% pull(suma)

  tibble(
    variable   = var_name,
    suma_tabla = suma_tabla,
    suma_pesos = suma_pesos,
    diferencia = suma_tabla - suma_pesos
  )
}

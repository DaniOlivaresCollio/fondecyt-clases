3.2.8 skills

  • Experts -> ciuo08_2d == 11 a 31 con universitaria completa o más (educ >= 9) —> experts

  • Skilled -> ciuo08_2d == 11 a 34 con universitaria incompleta o menos (educ < 9), y

  • Skilled -> ciuo08_2d == 35,36,41:44,53, y

  • Skilled -> ciuo08_2d == 51,54,61,62,71:75,81 con media completa o más (educ >= 5).

  • Unskilled -> ciuo08_2d == 51,54,61,62,71:75,81 con media incompleta o menos, y

  • Unskilled -> ciuo08_2d == 52,63,82,83,90+

  • 61 62 71:75 –> skilled

  • 51,54,81 –> unskilled

Ver código 
data <- data %>% 
  mutate(
    skills = case_when(
      # Experts
      ciuo08_ine %in% c(11:26) & educ >= 9                         ~ 1,
      ciuo08_ine < 11 & educ >= 9                                  ~ 1,

      # Skilled
      ciuo08_ine < 11 & (educ <= 8 | is.na(educ))                  ~ 2,
      ciuo08_ine %in% c(11:26) & (educ <= 8 | is.na(educ))         ~ 2,
      ciuo08_ine %in% c(31:36, 41:44, 53)                          ~ 2,
      ciuo08_ine %in% c(61, 62, 71:75) & (educ >= 5 | is.na(educ)) ~ 2,
      ciuo08_ine %in% c(51, 54, 81) & educ >= 5                    ~ 2,

      # Unskilled
      ciuo08_ine %in% c(61, 62, 71:75) & educ <= 4                 ~ 3,
      ciuo08_ine %in% c(51, 54, 81) & (educ <= 4 | is.na(educ))    ~ 3,
      ciuo08_ine %in% c(52, 63, 82, 83, 91:99)                     ~ 3,

      .default = NA_real_
    ),
    skills = factor(
      skills, 
      levels = 1:3,
      labels = c(
        "Experts",
        "Skilled",
        "Unskilled"
      ),
      ordered = TRUE
    )
  ) %>%
  set_variable_labels(skills = "Skills")

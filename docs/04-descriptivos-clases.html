prom_por_clase <- data %>%
  group_by(wrightclass2_reduced3) %>%
  summarise(
    mean_income = mean(income_rec, na.rm = TRUE),
    n           = n(),
    n_missing   = sum(is.na(income_rec)),
    .groups = "drop"

4.2.2 Ingresos x Clase ponderado

data_svy <- as_survey_design(data, weights = rake_wb2)

prom_por_clase_w <- data_svy %>%
  group_by(wrightclass2_reduced3) %>%
  summarise(
    mean_income_w = survey_mean(income_rec, na.rm = TRUE, vartype = NULL),
    n_w           = survey_total(!is.na(income_rec), na.rm = TRUE, vartype = NULL),
    missing_w     = survey_total(is.na(income_rec), vartype = NULL),
    .groups = "drop"
  )
